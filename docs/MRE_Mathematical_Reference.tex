\section{Notation and foundations}
{\color{mute}Symbols used throughout. Probabilities live in [0, 1]; evidence is added on the log-odds scale, measured in nats (natural-log units).}

\subsection*{Logit (log-odds)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{logit}(p)=\ln\frac{p}{1-p}$}\par\smallskip
Turns a probability into log-odds, where independent evidence adds up. Probabilities are clipped to [10\textasciicircum{}-6, 1-10\textasciicircum{}-6] (Stage 2) or [10\textasciicircum{}-9, 1-10\textasciicircum{}-9] before taking logs.
\where{everywhere}

\subsection*{Sigmoid (inverse logit)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \sigma(x)=\frac{1}{1+e^{-x}}$}\par\smallskip
Turns log-odds back into a probability. Arguments are clipped to [-40, 40] (or [-30, 30] in the encoders) to avoid overflow.
\where{everywhere}

\subsection*{Log-likelihood ratio (evidence weight)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{LLR}=\ln\frac{P(E\mid \mathrm{front})}{P(E\mid \mathrm{genuine})}$}\par\smallskip
How strongly a piece of evidence E favours "front" over "genuine", in nats. Positive supports a front, negative supports a genuine merchant, zero is neutral.
\where{frontend/src/engines/judge.ts}

\subsection*{Bayes rule on the log-odds scale}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{logit}\,P(\mathrm{front}\mid E)=\mathrm{logit}\,P(\mathrm{front})+\sum_i \mathrm{LLR}_i$}\par\smallskip
Posterior log-odds = prior log-odds + the sum of evidence weights (exact when the evidence is conditionally independent; the engine shrinks correlated evidence, see Part 11).
\where{frontend/src/engines/evidenceFusion.ts}

\subsection*{Share and excess share (effect sizes)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \hat{r}=\frac{x}{n},\qquad \mathrm{effect}=\hat{r}-r_0$}\par\smallskip
x events out of n, compared with a norm rate r0. Used by most rate-based detectors.
\where{frontend/src/engines/detectors.ts}

\section{Probability distributions}
{\color{mute}frontend/src/stats/dist.ts. Special functions (incomplete gamma and beta, erfc, log-gamma) come from the vetted @stdlib library; everything returns a probability clamped to [0, 1].}

\subsection*{Normal CDF, survival and quantile}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \Phi(z)=\frac{1}{2}\,\mathrm{erfc}\!\left(-\frac{z}{\sqrt{2}}\right),\qquad 1-\Phi(z)=\frac{1}{2}\,\mathrm{erfc}\!\left(\frac{z}{\sqrt{2}}\right)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \Phi^{-1}(p)=-\sqrt{2}\;\mathrm{erfc}^{-1}(2p)$}\par\smallskip
Standard normal probabilities via the complementary error function, accurate far into the tails.
\where{stats/dist.ts}

\subsection*{Chi-square survival function}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P(\chi^2_k\geq x)=Q\!\left(\frac{k}{2},\frac{x}{2}\right)=\frac{\Gamma(k/2,\,x/2)}{\Gamma(k/2)}$}\par\smallskip
Upper tail of a chi-square with k degrees of freedom, using the regularised upper incomplete gamma function Q. Returns 1 for x \textless{}= 0.
\where{stats/dist.ts}

\subsection*{Beta CDF and quantile}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle F_{\mathrm{Beta}}(x;a,b)=I_x(a,b),\qquad F^{-1}(p)=I^{-1}_p(a,b)$}\par\smallskip
Regularised incomplete beta function I and its inverse.
\where{stats/dist.ts}

\subsection*{Log binomial coefficient}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \ln\mathrm{C}(n,k)=\ln\Gamma(n+1)-\ln\Gamma(k+1)-\ln\Gamma(n-k+1)$}\par\smallskip
Computed with log-gamma so large n never overflows.
\where{stats/dist.ts}

\subsection*{Binomial upper tail (exact)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P(X\geq k)=I_p(k,\;n-k+1),\qquad X\sim\mathrm{Bin}(n,p)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P(X\leq k)=1-P(X\geq k+1)$}\par\smallskip
Exact binomial tails through the incomplete beta identity. Edge cases: k \textless{}= 0 gives 1, k \textgreater{} n gives 0.
\where{stats/dist.ts}

\subsection*{Poisson upper tail}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P(X\geq k)=P(k,\lambda)=\frac{\gamma(k,\lambda)}{\Gamma(k)},\qquad X\sim\mathrm{Poisson}(\lambda)$}\par\smallskip
Exact Poisson tail through the regularised lower incomplete gamma function.
\where{stats/dist.ts}

\subsection*{Hypergeometric upper tail}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P(X\geq k)=\sum_{i=k}^{\min(n,K)}\frac{\mathrm{C}(K,i)\,\mathrm{C}(N-K,\,n-i)}{\mathrm{C}(N,n)}$}\par\smallskip
Drawing n items from N of which K are successes, without replacement. Summed in log space for stability.
\where{stats/dist.ts}

\subsection*{Beta-binomial (over-dispersed rates)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \ln P(X=k)=\ln\mathrm{C}(n,k)+\ln B(k+a,\;n-k+b)-\ln B(a,b)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \ln B(a,b)=\ln\Gamma(a)+\ln\Gamma(b)-\ln\Gamma(a+b),\qquad P(X\geq k)=\sum_{i=k}^{n}P(X=i)$}\par\smallskip
The rate itself varies between merchants (Beta(a, b)), so counts are more spread out than a plain binomial. The engine's main rate test.
\where{stats/dist.ts}

\subsection*{Negative binomial}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \ln P(X=k)=\ln\Gamma(k+\theta)-\ln\Gamma(\theta)-\ln\Gamma(k+1)+\theta\ln p+k\ln(1-p),\quad p=\frac{\theta}{\theta+\mu}$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{Var}(X)=\mu+\frac{\mu^2}{\theta},\qquad [\mathrm{lo},\mathrm{hi}]=\mu\mp z_{1-\alpha/2}\sqrt{\mu+\mu^2/\theta}$}\par\smallskip
Over-dispersed counts with mean mu and size theta; the prediction interval uses a normal approximation on the NB moments (floored at 0).
\where{stats/dist.ts}

\subsection*{Log-normal CDF}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle F(x)=\Phi\!\left(\frac{\ln(x/m)}{\sigma}\right),\qquad x>0$}\par\smallskip
Ticket sizes: m is the category median ticket, sigma the log-scale spread.
\where{stats/dist.ts}

\section{Hypothesis tests}
{\color{mute}frontend/src/stats/tests.ts. Every test returns a statistic and a p-value.}

\subsection*{G-test of goodness of fit}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle G=2\sum_i O_i\ln\frac{O_i}{E_i},\qquad E_i=n\,\frac{p_i}{\sum_j p_j},\qquad p=P(\chi^2_{\mathrm{df}}\geq G)$}\par\smallskip
Cells with expected count below 5 are pooled into one "other" cell; df = cells - 1 (at least 1).
\where{stats/tests.ts}

\subsection*{Jensen-Shannon divergence (effect size)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{JSD}(P\|Q)=\frac{1}{2}D_{KL}(P\|M)+\frac{1}{2}D_{KL}(Q\|M),\quad M=\frac{P+Q}{2},\quad D_{KL}(A\|B)=\sum_i a_i\log_2\frac{a_i}{b_i}$}\par\smallskip
Distance between observed and expected shares, in bits, bounded in [0, 1].
\where{stats/tests.ts}

\subsection*{Two-sample Kolmogorov-Smirnov}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle D=\max_x\left|F_a(x)-F_b(x)\right|,\qquad n_e=\frac{n_a n_b}{n_a+n_b}$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \lambda=\left(\sqrt{n_e}+0.12+\frac{0.11}{\sqrt{n_e}}\right)D,\qquad p=2\sum_{k=1}^{100}(-1)^{k-1}e^{-2k^2\lambda^2}$}\par\smallskip
Largest gap between two empirical CDFs; asymptotic p-value with Stephens' small-sample correction.
\where{stats/tests.ts}

\subsection*{One-sample Kolmogorov-Smirnov}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle D=\max_i\max\left(\frac{i}{n}-F(x_{(i)}),\;F(x_{(i)})-\frac{i-1}{n}\right),\qquad \lambda=\left(\sqrt{n}+0.12+\frac{0.11}{\sqrt{n}}\right)D$}\par\smallskip
Sample against a continuous reference CDF (used by detector A1 against the log-normal ticket shape); same series for p.
\where{stats/tests.ts}

\subsection*{Exact one-sided binomial test}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p=P(X\geq x\mid n,p_0),\qquad \mathrm{statistic}=x/n$}\par\smallskip
Is x successes out of n more than a rate p0 explains?
\where{stats/tests.ts}

\subsection*{Poisson rate-ratio (exact conditional)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle c_1\mid c_1+c_2\sim\mathrm{Bin}\!\left(c_1+c_2,\;\frac{t_1}{t_1+t_2}\right),\qquad \mathrm{ratio}=\frac{c_1/t_1}{\max(c_2/t_2,\;0.5/t_2)}$}\par\smallskip
Compares two event rates (counts c over exposure t); one-sided p = P(X \textgreater{}= c1).
\where{stats/tests.ts}

\subsection*{Permutation test}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p=\frac{1+\#\{b:\;T(\pi_b)\geq T_{\mathrm{obs}}\}}{B+1}$}\par\smallskip
Shuffle group labels B times (2,000 by default) and count statistics at least as extreme. Exact and never zero. Used by detector C5.
\where{stats/tests.ts}

\subsection*{Mann-Whitney U (tie-corrected)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle U=R_1-\frac{n_1(n_1+1)}{2},\qquad \mu_U=\frac{n_1 n_2}{2}$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \sigma_U=\sqrt{\frac{n_1 n_2}{12}\left(N+1-\frac{\sum_j(t_j^3-t_j)}{N(N-1)}\right)},\qquad p=1-\Phi\!\left(\frac{\mu_U-U-0.5}{\sigma_U}\right)$}\par\smallskip
Rank test that values in one group tend to be smaller; t\_j are tie group sizes; 0.5 is a continuity correction.
\where{stats/tests.ts}

\subsection*{Fisher's method}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle X=-2\sum_{i=1}^{m}\ln p_i\;\sim\;\chi^2_{2m}$}\par\smallskip
Combines independent p-values.
\where{stats/tests.ts}

\subsection*{Cauchy combination (Liu and Xie 2020)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle T=\sum_i w_i\tan\!\left((0.5-p_i)\pi\right),\qquad p=\frac{1}{2}-\frac{\arctan T}{\pi}$}\par\smallskip
Combines p-values validly under arbitrary dependence; weights default to 1/m.
\where{stats/tests.ts}

\subsection*{Exact McNemar test}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p=\min\!\left(1,\;2\,P\!\left(X\geq\max(b,c)\right)\right),\qquad X\sim\mathrm{Bin}(b+c,\;0.5)$}\par\smallskip
Paired decisions: b = cases model A gets right and B wrong, c = the reverse.
\where{stats/tests.ts, ml/stage2\_math.py}

\subsection*{Sellke-Bayarri-Berger bound}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{LLR}_{\max}=-\ln\!\left(-e\,p\ln p\right)\quad (p<1/e),\qquad \mathrm{capped\ at\ 4\ nats}$}\par\smallskip
The largest Bayes factor a p-value can support; a conservative evidence weight until a detector's likelihood ratio has been learned.
\where{stats/tests.ts}

\section{Empirical Bayes rate model}
{\color{mute}frontend/src/stats/eb.ts and detectors.ts. Small merchants are shrunk towards their peers instead of being flagged on a handful of sales.}

\subsection*{Method-of-moments Beta fit to peer rates}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \hat{p}=\frac{\sum x_i}{\sum n_i},\qquad \bar{n}=\frac{\sum n_i}{k},\qquad s^2=\frac{\sum_i n_i\,(x_i/n_i-\hat{p})^2}{(k-1)\,\bar{n}}$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \tau^2=\max\!\left(s^2-\frac{\hat{p}(1-\hat{p})}{\bar{n}},\;10^{-6}\right),\qquad a+b=\mathrm{clip}\!\left(\frac{\hat{p}(1-\hat{p})}{\tau^2}-1,\;2,\;10^5\right)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle a=\hat{p}\,(a+b),\qquad b=(1-\hat{p})\,(a+b)$}\par\smallskip
Fits Beta(a, b) to peers, subtracting binomial sampling noise from the between-merchant variance. Needs at least 3 peers, else Beta(1, 1).
\where{stats/eb.ts}

\subsection*{Prior from a norm rate}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle a=\max(r\,s,\;10^{-3}),\qquad b=\max((1-r)\,s,\;10^{-3})$}\par\smallskip
A category norm rate r with strength s "worth s merchant-sales". Detectors use s = 200 (100 for H3).
\where{stats/eb.ts}

\subsection*{Shrunk (posterior mean) rate}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \tilde{r}=\frac{x+a}{n+a+b}$}\par\smallskip
The merchant's rate pulled towards the prior.
\where{stats/eb.ts}

\subsection*{Rate test used by the detectors}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p=P(X\geq x),\quad X\sim\mathrm{BetaBin}(n,a,b);\qquad \mathrm{effect}=\frac{x}{n}-r_0;\qquad \mathrm{CI}=\mathrm{Wilson}_{90\%}(x,n)$}\par\smallskip
Upper-tail beta-binomial p-value against the category norm, the excess share as effect size, and a 90\% Wilson interval.
\where{engines/detectors.ts}

\section{Confidence intervals}
{\color{mute}frontend/src/stats/intervals.ts (also ml/stage2\_math.py and engine/improver.ts).}

\subsection*{Wilson score interval}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \frac{\hat{p}+\frac{z^2}{2n}\;\pm\;z\sqrt{\frac{\hat{p}(1-\hat{p})}{n}+\frac{z^2}{4n^2}}}{1+\frac{z^2}{n}},\qquad z=\Phi^{-1}\!\left(1-\frac{1-\mathrm{level}}{2}\right)$}\par\smallskip
Reliable interval for a proportion, even near 0 or 1; z = 1.96 at 95\%, 1.645 at 90\%.
\where{stats/intervals.ts}

\subsection*{Clopper-Pearson (exact) interval}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \left[\,I^{-1}_{\alpha/2}(x,\;n-x+1),\;\;I^{-1}_{1-\alpha/2}(x+1,\;n-x)\,\right]$}\par\smallskip
Exact binomial interval from beta quantiles (0 when x = 0, 1 when x = n).
\where{stats/intervals.ts}

\subsection*{Newcombe interval for a difference}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle d=\hat{p}_1-\hat{p}_2,\qquad \left[d-\sqrt{(\hat{p}_1-l_1)^2+(u_2-\hat{p}_2)^2},\;\;d+\sqrt{(u_1-\hat{p}_1)^2+(\hat{p}_2-l_2)^2}\right]$}\par\smallskip
Hybrid score interval for p1 - p2, built from the two Wilson intervals (l, u).
\where{stats/intervals.ts}

\subsection*{Interpolated quantile}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle Q(q)=x_{(\lfloor h\rfloor)}+\left(h-\lfloor h\rfloor\right)\left(x_{(\lceil h\rceil)}-x_{(\lfloor h\rfloor)}\right),\qquad h=(n-1)\,q$}\par\smallskip
Linear interpolation between order statistics.
\where{stats/intervals.ts}

\subsection*{Percentile bootstrap}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{CI}=\left[Q_{\alpha/2}(\hat{\theta}^{*}_1..\hat{\theta}^{*}_B),\;Q_{1-\alpha/2}(\hat{\theta}^{*}_1..\hat{\theta}^{*}_B)\right]$}\par\smallskip
Resample the data with replacement B times (1,000 by default) and take quantiles of the recomputed statistic. Stage 2 skips resamples missing a class.
\where{stats/intervals.ts, ml/stage2\_lib.py}

\section{Multiple testing}
{\color{mute}frontend/src/stats/multiple.ts and the Stage 2 drift endpoint (ml/serve.py).}

\subsection*{Holm step-down (family-wise error)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \tilde{p}_{(r)}=\max_{j\leq r}\;\min\!\left(1,\;(m-j+1)\,p_{(j)}\right)$}\par\smallskip
Adjusted p-values in sorted order, made monotone.
\where{stats/multiple.ts}

\subsection*{Benjamini-Hochberg q-values (false discovery rate)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle q_{(r)}=\min_{j\geq r}\;\min\!\left(1,\;\frac{m\,p_{(j)}}{j}\right),\qquad \mathbb{E}[\mathrm{false\ discoveries}]\approx q\,R$}\par\smallskip
Reject hypotheses with q \textless{}= target; R is the number rejected.
\where{stats/multiple.ts, ml/serve.py}

\subsection*{Multiple-stream log-odds penalty}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \Delta\,\mathrm{logit}=-\ln m$}\par\smallskip
When a merchant's m QR links and repeat-payer group are each scored separately, every stream's prior is lowered by ln m (a Bonferroni-style correction on the log-odds scale).
\where{engines/investigationEngine.ts}

\section{Seeded randomness}
{\color{mute}frontend/src/stats/rng.ts and utils/prng.ts. Same seed, same result, so every stochastic routine is reproducible.}

\subsection*{FNV-1a string seed}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle h_0=2166136261,\qquad h_{i+1}=\left(h_i\oplus c_i\right)\times 16777619\ \ \mathrm{mod}\ 2^{32}$}\par\smallskip
A stable 32-bit seed from a merchant id, so each merchant's random draws are reproducible.
\where{stats/rng.ts}

\subsection*{Box-Muller normal draw}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle Z=\sqrt{-2\ln U_1}\,\cos(2\pi U_2),\qquad U_1,U_2\sim\mathrm{U}(0,1)$}\par\smallskip
Standard normal from two uniforms.
\where{stats/rng.ts, engine/synthetic.ts}

\subsection*{Marsaglia-Tsang gamma sampler}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle d=a-\frac{1}{3},\quad c=\frac{1}{\sqrt{9d}},\quad v=(1+cZ)^3,\quad \mathrm{accept\ }dv\ \mathrm{if}\ \ln U<\frac{Z^2}{2}+d\,(1-v+\ln v)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{Gamma}(a<1)=\mathrm{Gamma}(a+1)\cdot U^{1/a},\qquad \mathrm{Beta}(a,b)=\frac{G_a}{G_a+G_b}$}\par\smallskip
Fast exact gamma draws (with the quick squeeze test U \textless{} 1 - 0.0331 Z\textasciicircum{}4); beta draws from two gammas.
\where{stats/rng.ts}

\subsection*{Fisher-Yates shuffle and bootstrap resample}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{for}\ i=n-1..1:\ j=\lfloor U(i+1)\rfloor,\ \mathrm{swap}(a_i,a_j);\qquad x^*_i=x_{\lfloor Un\rfloor}$}\par\smallskip
Uniform random permutation (permutation tests) and sampling with replacement (bootstrap).
\where{stats/rng.ts}

\section{Volume model and change detection}
{\color{mute}frontend/src/engines/residualEngine.ts, cusumEngine.ts and stats/changepoint.ts.}

\subsection*{Festival-aware expected weekly volume}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathbb{E}[V_t]=\mathrm{base}\times m_t,\qquad m_t=m^{\mathrm{festival}}_t\times\mathrm{uplift}^{\mathrm{passport}}_t$}\par\smallskip
Festival multipliers come from the calendar ramp for the merchant's sector (never from the merchant's own upload); a verified Context Passport can add an uplift capped at 3x.
\where{engines/residualEngine.ts}

\subsection*{Baseline from clean weeks}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \bar{\ell}=\frac{1}{n}\sum_{t\in\mathcal{B}}\ln V_t,\qquad \mathrm{base}=e^{\bar{\ell}}$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle cv=\max\!\left(0.05,\;\sqrt{\frac{1}{n-1}\sum_{t\in\mathcal{B}}(\ln V_t-\bar{\ell})^2}\right)$}\par\smallskip
B = non-festival weeks before the tested window (up to 8 weeks; at least 3 needed, else the declared baseline and cv = 0.15).
\where{engines/residualEngine.ts}

\subsection*{Standardised log residual}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle z_t=\frac{\ln V_t-\ln\mathbb{E}[V_t]}{\sqrt{cv^2+\sigma_f^2\,\mathbb{1}[m_t>1]}},\qquad \sigma_f=0.35$}\par\smallskip
Festival weeks are judged more loosely, because their multiplier is itself uncertain.
\where{engines/residualEngine.ts}

\subsection*{Relative residuals (displayed)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle r^{\mathrm{anch}}_t=\frac{V_t-\mathrm{base}}{\max(\mathrm{base},1)},\quad r^{\mathrm{seas}}_t=\frac{V_t-\mathbb{E}[V_t]}{\max(\mathbb{E}[V_t],1)},\quad r^{\mathrm{roll}}_t=\frac{V_t-\bar{V}_{t-4..t-1}}{\max(\bar{V},1)}$}\par\smallskip
Actual against the anchored baseline, the seasonal expectation and a 4-week rolling mean.
\where{engines/residualEngine.ts}

\subsection*{One-sided CUSUM}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle S_t=\max(0,\;S_{t-1}+z_t-k),\qquad \mathrm{alarm\ if}\ S_t>h,\qquad k=0.5,\;h=2.35$}\par\smallskip
Accumulates sustained upward drift in the residuals.
\where{engines/cusumEngine.ts, stats/changepoint.ts}

\subsection*{CUSUM threshold calibration (ARL)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle h^{*}=\min\{h:\;\mathrm{ARL}_0(k,h)\geq 52\ \mathrm{weeks}\},\qquad \mathrm{ARL}_0=\frac{1}{S}\sum_{s=1}^{S}\mathrm{run\ length}_s$}\par\smallskip
Simulate N(0,1) data (400 runs, cap 5,000) and bisect h (18 steps, rounded up to 0.05) so a well-behaved merchant raises one false alarm per year.
\where{stats/changepoint.ts}

\subsection*{Bayesian online changepoint detection (Adams and MacKay 2007)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \pi_t(x)=\mathcal{N}\!\left(x;\;\mu_r,\;\sigma^2_r+1\right),\qquad R_{t+1}(r+1)\propto R_t(r)\,\pi_t\,(1-H),\qquad R_{t+1}(0)\propto\sum_r R_t(r)\,\pi_t\,H$}\par\smallskip

\where{engines/judge.ts}

\subsection*{UI evidence weight}

\section{Bayesian evidence fusion}
{\color{mute}frontend/src/engines/evidenceFusion.ts. Combines all detectors into P(front) with an uncertainty interval, then applies the governance gate.}

\subsection*{Prior}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{logit}\,\pi=\mathrm{logit}(r_{\mathrm{MCC}})+\ln(\mathrm{odds}_{\mathrm{age}})+\Delta$}\par\smallskip
Category base rate (default 1.5\%), multiplied by account-age odds (2.0 under 90 days, 1.3 under a year, 1.0 under 3 years, 0.7 after), plus any stream penalty Delta.
\where{engines/evidenceFusion.ts, rules/priors.json}

\subsection*{Family LLR with correlation shrinkage}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{LLR}_{F}=\ell^{+}_{(1)}+\rho\sum_{j\geq 2}\ell^{+}_{(j)}\;+\;\ell^{-}_{(1)}+\rho\sum_{j\geq 2}\ell^{-}_{(j)},\qquad \rho=0.3$}\par\smallskip
Detectors in the same family share data, so only the strongest positive and strongest negative count fully; the rest are multiplied by rho.
\where{engines/evidenceFusion.ts}

\subsection*{Diminishing returns across families}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{LLR}_{F_{(i)}}\leftarrow 0.85\,\mathrm{LLR}_{F_{(i)}}\ \ \mathrm{for\ positive\ families\ ranked}\ i>3$}\par\smallskip
Beyond the three strongest supporting families, each counts 85\%.
\where{engines/evidenceFusion.ts}

\subsection*{Posterior}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{logit}\,P(\mathrm{front}\mid E)=\mathrm{logit}\,\pi+\sum_{F}\mathrm{LLR}_F+\sum\mathrm{overrides},\qquad P=\sigma(\cdot)$}\par\smallskip
Hard overrides only add odds: R-01 (link on a known betting host, +3 nats, needs effect \textgreater{}= 1.25) and R-02 (shared beneficiary with a handed-off peer, +2 nats).
\where{engines/evidenceFusion.ts}

\subsection*{Monte-Carlo 90\% credible interval}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \tilde{s}=\sigma\!\left(\mathrm{logit}\,s+\epsilon\,\sigma_s\right),\quad \sigma_s=\frac{1}{\sqrt{1+k\,s(1-s)}},\quad k=20,\quad \epsilon\sim\mathcal{N}(0,1)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \tilde{\ell}_i=\max\!\left(-1.5,\;\ell_i\,\frac{\mathrm{LLR}(\tilde{s},\tilde{f})}{\mathrm{LLR}(s,f)}\right),\qquad \tilde{r}=\sigma\!\left(\mathrm{logit}\,r+\epsilon\,\frac{1}{\sqrt{1+400\,r(1-r)}}\right)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{CI}_{90}=\left[Q_{0.05}(P^{(d)}),\;Q_{0.95}(P^{(d)})\right],\qquad d=1..1000$}\par\smallskip
Each detector's sensitivity and false-positive rate, and the base rate, are drawn from logit-normal distributions whose width reflects how much data stands behind them; the whole fusion is repeated 1,000 times.
\where{engines/evidenceFusion.ts}

\subsection*{Governance gate (case opens only if all hold)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P\geq 0.10\ \ \wedge\ \ \#\mathrm{conduct}>0\ \ \wedge\ \ \left(\#\mathrm{sources}_{\geq 0.5}\geq 2\ \vee\ \#\mathrm{families}_{\geq 0.5}\geq 3\ \vee\ \#\mathrm{behaviours}\geq 2\right)$}\par\smallskip
Independence: evidence of at least 0.5 nats from two data sources, or three families, or two different conduct behaviours. Volume (H), the declaration (K) and KYC do not count towards independence.
\where{engines/evidenceFusion.ts, rules/engine.json}

\subsection*{Question instead of a case (ASK)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{ASK}\Leftrightarrow\neg\mathrm{gate}\ \wedge\ \left(\#\mathrm{conduct}>0\ \vee\ (P\geq 0.01\ \wedge\ \#\mathrm{fired\ families}\geq 2)\right)$}\par\smallskip
A departure from normal alone produces a Context Passport question, never a restriction.
\where{engines/evidenceFusion.ts}

\subsection*{Mechanism posterior (softmax)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P(\mu\mid\mathrm{front})=\frac{\pi_\mu\exp\left(\sum_i\ell^{+}_i\,a_{i\mu}\right)}{\sum_{\nu}\pi_\nu\exp\left(\sum_i\ell^{+}_i\,a_{i\nu}\right)},\qquad \mathrm{score}_\mu=P\cdot P(\mu\mid\mathrm{front})$}\par\smallskip
Which kind of abuse: laundering (prior 0.40), rotation (0.20), fabricated substance (0.25), other (0.15); a\_i,mu = each detector's affinity.
\where{engines/evidenceFusion.ts}

\subsection*{Interval-based decision set and priority}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathcal{C}=\{\mathrm{GENUINE}\}\ \mathrm{if}\ u<\tau;\ \ \{\mathrm{FRONT}\}\ \mathrm{if}\ l\geq\tau;\ \ \mathrm{else\ both},\qquad \tau=0.10$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{priority}=\mathrm{HIGH}\ (P\geq 0.8),\ \ \mathrm{MEDIUM}\ (P\geq 0.5),\ \ \mathrm{LOW\ otherwise}$}\par\smallskip
l and u are the 90\% interval ends; when the interval straddles the bar, abstaining is recommended.
\where{engines/evidenceFusion.ts}

\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{HHI}=\sum_i\left(\frac{c_i}{n}\right)^2,\qquad \mathrm{CV}_{\mathrm{ticket}}=\frac{\sqrt{\frac{1}{n}\sum(x-\bar{x})^2}}{\bar{x}}$}\par\smallskip
Herfindahl-Hirschman index of payer shares; coefficient of variation of ticket amounts.
\where{engine/features.ts}

\subsection*{Sweep lag}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{lag}_s=t_{\mathrm{next\ payout}}-t_s\ \ (0\leq\mathrm{lag}\leq 72\ \mathrm{h}),\qquad \mathrm{sweep\ lag}=\mathrm{median}(\mathrm{lag}_s)\ \ (72\ \mathrm{if\ none})$}\par\smallskip
How fast money leaves after each settlement, found by binary search.
\where{engine/features.ts}

\subsection*{Benford's law (first digit)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P(d)=\log_{10}\!\left(1+\frac{1}{d}\right),\ d=1..9,\qquad \chi^2=\sum_d\frac{(O_d-nP(d))^2}{nP(d)},\ \ p=P(\chi^2_8\geq\chi^2)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{MAD}=\frac{1}{9}\sum_d\left|\frac{O_d}{n}-P(d)\right|,\qquad \chi^2/n,\qquad d=\left\lfloor x/10^{\lfloor\log_{10}x\rfloor}\right\rfloor$}\par\smallskip
Natural amounts follow Benford's first-digit law; MAD is Nigrini's conformity measure (about 0.006 close, 0.015 nonconforming). Neutral values below 30 sales.
\where{engine/fraudMath.ts}

\subsection*{Payment timing}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{dispersion}=\frac{\mathrm{Var}(\mathrm{sales/day})}{\mathrm{Mean}(\mathrm{sales/day})},\qquad B=\frac{\sigma_g-\mu_g}{\sigma_g+\mu_g},\qquad \mathrm{CV}_g=\frac{\sigma_g}{\mu_g}$}\par\smallskip
Index of dispersion (1 for a Poisson process) and Goh-Barabasi burstiness of inter-arrival gaps (-1 periodic, 0 random, towards 1 bursty).
\where{engine/fraudMath.ts}

\subsection*{Payer entropy and Gini}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle H_{\mathrm{norm}}=-\frac{\sum_i p_i\ln p_i}{\ln K},\qquad G=\frac{2\sum_{i=1}^{K}i\,x_{(i)}}{K\sum_i x_i}-\frac{K+1}{K}$}\par\smallskip
Normalised Shannon entropy (1 = every payer equally often) and the Gini coefficient of payer counts.
\where{engine/fraudMath.ts}

\subsection*{Repeat-payer patterns}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{returning\ share}=\frac{\#\mathrm{sales\ from\ payers\ seen}\geq 2\ \mathrm{times}}{n},\qquad \mathrm{repeat\ gap\ CV}=\mathrm{mean}_{\mathrm{payer}}\frac{\sigma_{\mathrm{gaps}}}{\mu_{\mathrm{gaps}}}$}\par\smallskip
How much business comes from returning payers, and how regular their returns are.
\where{engine/fraudMath.ts}

\section{Stage 2 models and training}
{\color{mute}ml/train\_stage2.py, ml/stage2\_lib.py, ml/focal.py and ml/deep\_ensemble.py.}

\subsection*{Logistic regression}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p=\sigma\!\left(\beta_0+\sum_j\beta_j z_j\right),\qquad z_j=\frac{x_j-\mu_j}{\sigma_j},\qquad \min\ -\sum_i w_i\ell_i+\frac{1}{2C}\|\beta\|^2$}\par\smallskip
Standardised inputs, balanced class weights, C chosen by cross-validation.
\where{ml/train\_stage2.py}

\subsection*{Decision tree (Gini impurity)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{Gini}=1-\sum_k p_k^2,\qquad \mathrm{split\ maximises}\ \ \mathrm{Gini}(\mathrm{parent})-\sum_{c}\frac{n_c}{n}\mathrm{Gini}(c)$}\par\smallskip
Shallow tree (depth 3-5).
\where{ml/train\_stage2.py}

\subsection*{Cost-complexity pruning (opt-in)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle R_\alpha(T)=R(T)+\alpha\,|\mathrm{leaves}(T)|$}\par\smallskip
Breiman et al. 1984; candidate alpha values from the tree's own pruning path. Enabled with MRE\_EXTENDED\_TREES=1.
\where{ml/train\_stage2.py}

\subsection*{Random forest}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p(x)=\frac{1}{B}\sum_{b=1}^{B}T_b(x),\qquad B=300$}\par\smallskip
Trees on bootstrap samples with sqrt(features) per split and balanced sub-sample weights.
\where{ml/train\_stage2.py}

\subsection*{Gradient-boosted trees (LightGBM)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle F_m(x)=F_{m-1}(x)+\eta\,h_m(x),\qquad p=\sigma(F_M(x)),\qquad \partial p/\partial x_j\geq 0\ \ \mathrm{for\ llr_\ast,\ fired_\ast}$}\par\smallskip
Boosting with learning rate eta = 0.05 and monotone constraints, so more evidence can never lower the score.
\where{ml/train\_stage2.py}

\subsection*{Focal loss (opt-in)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle t=2y-1,\quad p_t=\sigma(tz),\qquad \mathrm{FL}=-(1-p_t)^\gamma\ln p_t$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle g=t\left[\gamma\,p_t(1-p_t)^\gamma\ln p_t-(1-p_t)^{\gamma+1}\right]$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle h=p_t(1-p_t)\left\{\gamma\left[(1-p_t)^\gamma(\ln p_t+1)-\gamma\,p_t(1-p_t)^{\gamma-1}\ln p_t\right]+(\gamma+1)(1-p_t)^\gamma\right\}$}\par\smallskip
Lin et al. 2017; gradient g and Hessian h with respect to the raw score z (checked against finite differences). gamma = 0 reduces to log loss (g = p - y, h = p(1 - p)).
\where{ml/focal.py}

\subsection*{Neural network (MLP 32x16)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle h_1=\mathrm{ReLU}(W_1z+b_1),\quad h_2=\mathrm{ReLU}(W_2h_1+b_2),\quad p=\sigma(w_3^\top h_2+b_3),\qquad \mathrm{ReLU}(x)=\max(0,x)$}\par\smallskip
L2 penalty alpha chosen by cross-validation; class balance through sample weights.
\where{ml/train\_stage2.py}

\subsection*{Deep ensemble (opt-in)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p(x)=\frac{1}{M}\sum_{m=1}^{M}p_m(x),\qquad M=5,\qquad \mathrm{Var}_m\left[\mathrm{logit}\,p_m(x)\right]$}\par\smallskip
Lakshminarayanan et al. 2017; members differ only by random seed. Enabled with MRE\_DEEP\_ENSEMBLE=1.
\where{ml/deep\_ensemble.py}

\subsection*{Coverage-dropout augmentation}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle x^{\mathrm{aug}}_{d}=0\ \ \mathrm{with\ probability}\ 0.4,\quad d\in\{\mathrm{B9,D7,D9,D10,E1,E3,F2,F5,F7,F8,G2,G6,C5}\}$}\par\smallskip
A copy of each training merchant with optional-feed detectors blanked, so a model cannot learn "feed present = honest".
\where{ml/stage2\_lib.py}

\subsection*{Grouped cross-validation and log-loss selection}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{LL}=-\frac{1}{\sum w_i}\sum_i w_i\left[y_i\ln p_i+(1-y_i)\ln(1-p_i)\right]$}\par\smallskip
StratifiedGroupKFold (5 folds, group = merchant id); every hyperparameter is chosen by weighted log loss, a proper scoring rule.
\where{ml/train\_stage2.py}

\subsection*{Analyst labels and prevalence}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle w_{\mathrm{label}}=5,\quad w_{\mathrm{synthetic}}=1,\qquad \pi_{\mathrm{train}}=\frac{\sum w_iy_i}{\sum w_i}$}\par\smallskip
Real analyst labels count five times a synthetic merchant.
\where{ml/train\_stage2.py}

\section{Stage 2 ensemble, calibration and decisions}
{\color{mute}ml/train\_stage2.py, ml/stage2\_math.py and ml/serve.py.}

\subsection*{Stacked generalisation (Wolpert 1992)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p=\sigma\!\left(a+\sum_k w_k\,\mathrm{logit}\,p_k\right),\qquad w_k\geq 0$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \min_{a,w}\;-\frac{1}{\sum v_i}\sum_i v_i\left[y_i\ln p_i+(1-y_i)\ln(1-p_i)\right]+0.01\|w\|^2$}\par\smallskip
A meta-model over the five base models' out-of-fold logits, fitted with L-BFGS-B; non-negative weights forbid exploiting an anti-correlated model.
\where{ml/train\_stage2.py}

\subsection*{Platt scaling (v0 models)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle q=\sigma\!\left(A\,\mathrm{logit}(p)+B\right)$}\par\smallskip
Logistic regression on out-of-fold logits, stored as a 21-point lookup grid.
\where{ml/train\_m1.py, ml/train\_nn.py}

\subsection*{Beta calibration (Kull et al. 2017)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{logit}\,q=a\ln p-b\ln(1-p)+c,\qquad a,b\geq 0$}\par\smallskip
Contains Platt scaling (a = b) and the identity (a = b = 1, c = 0).
\where{ml/stage2\_math.py}

\subsection*{Isotonic calibration (pool-adjacent-violators)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \hat{q}=\arg\min_{q\ \mathrm{non\text{-}decreasing}}\sum_i w_i(y_i-q(p_i))^2$}\par\smallskip
Best monotone fit; the calibrator (identity, beta or isotonic) is chosen by cross-fitted log loss. Identity currently wins.
\where{ml/stage2\_math.py}

\subsection*{Prior-shift correction (Saerens et al. 2002)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{logit}\,p^\prime=\mathrm{logit}\,p-\mathrm{logit}\,\pi_{\mathrm{train}}+\mathrm{logit}\,\pi_{\mathrm{engine}}$}\par\smallskip
Moves the probability from the training base rate (about 34\%) onto the engine's prior for that merchant (about 1-1.5\%).
\where{ml/serve.py}

\subsection*{Bayes cost threshold (Elkan 2001)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{flag}\Leftrightarrow p^\prime\geq\frac{C_{FP}}{C_{FP}+C_{FN}}=\frac{1}{1+20}\approx 0.048$}\par\smallskip
A missed front counted as 20 needless reviews. On the training scale: logit t = logit(1/21) - logit(pi\_engine) + logit(pi\_train).
\where{ml/train\_stage2.py}

\subsection*{Neyman-Pearson threshold (Tong, Feng and Li 2018)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle k^{*}=\min\left\{k:\;\sum_{j=k}^{n}\mathrm{C}(n,j)(1-\alpha)^j\alpha^{n-j}\leq\delta\right\},\qquad t=s_{(k^{*})}$}\par\smallskip
From the sorted honest-merchant scores; guarantees P(false-alarm rate \textgreater{} alpha) \textless{}= delta (alpha = delta = 0.05). Needs n \textgreater{}= ln(delta) / ln(1 - alpha) = 59 honest merchants.
\where{ml/stage2\_math.py}

\subsection*{Expected cost curve}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{EC}(t)=C_{FN}\,\pi\,(1-\mathrm{TPR}(t))+C_{FP}\,(1-\pi)\,\mathrm{FPR}(t)$}\par\smallskip
Re-weighted to the deployment prevalence pi.
\where{ml/stage2\_math.py}

\subsection*{Mondrian split-conformal prediction (Vovk et al. 2005)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle s_1=1-p\ (\mathrm{fronts}),\quad s_0=p\ (\mathrm{honest}),\qquad q_c=\mathrm{Quantile}\!\left(s_c;\;\frac{\lceil(n_c+1)(1-\alpha)\rceil}{n_c}\right)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p_c=\frac{\#\{s_i\geq s\}+1}{n_c+1},\qquad \mathcal{C}=\{c:\ p_c>\alpha\},\qquad \alpha=0.1$}\par\smallskip
A 90\% prediction set per class. \{front\} or \{honest\} is a confident call; both or neither means "unsure". Class-conditional validity survives the prior shift.
\where{ml/train\_stage2.py, ml/serve.py}

\subsection*{Smoothed conformal p-value}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p=\frac{\#\{s_i>s\}+U\,(\#\{s_i=s\}+1)}{n+1},\qquad U\sim\mathrm{U}(0,1)$}\par\smallskip
Exactly uniform under exchangeability.
\where{ml/stage2\_math.py}

\subsection*{Novelty detection (Mahalanobis + Ledoit-Wolf)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle z=\mathrm{clip}\!\left(\Phi^{-1}(\hat{F}_j(x_j)),\;-5.2,\;5.2\right),\qquad \hat{\Sigma}=(1-\lambda)S+\lambda F$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle d(x)=\sqrt{(z-\bar{z})^\top\hat{\Sigma}^{-1}(z-\bar{z})},\qquad \mathrm{novel}\Leftrightarrow d>Q_{0.995}(d^{\mathrm{OOF}})$}\par\smallskip
Quantile-normal transform per feature, shrunk covariance, and a threshold set on out-of-fold distances. Drivers: largest z\_j (Sigma\textasciicircum{}-1 z)\_j terms.
\where{ml/stage2\_lib.py}

\subsection*{Novelty p-value}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p_{\mathrm{nov}}=\frac{\#\{d^{\mathrm{OOF}}_i\geq d\}+1}{n+1}$}\par\smallskip
How unusual a merchant is compared with training.
\where{ml/serve.py}

\subsection*{Entropy decomposition of uncertainty}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle H(p)=-p\log_2 p-(1-p)\log_2(1-p)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle H_{\mathrm{total}}=H\!\left(\sum_k w_kp_k\right),\qquad H_{\mathrm{aleatoric}}=\sum_k w_kH(p_k)$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle H_{\mathrm{total}}=H_{\mathrm{aleatoric}}+I(y;k),\qquad H_{\mathrm{epistemic}}=I(y;k)\geq 0$}\par\smallskip
Depeweg et al. 2018 / BALD: noise the members agree on vs disagreement between members, in bits.
\where{ml/stage2\_math.py}

\subsection*{Confidence rule}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{LOW}\ \mathrm{if\ novel}\ \vee\ |\mathcal{C}|\neq 1;\quad \mathrm{MEDIUM\ if}\ I(y;k)>0.1\ \mathrm{bit};\quad \mathrm{HIGH\ otherwise}$}\par\smallskip
\where{ml/serve.py}

\subsection*{Stage 1 + Stage 2 blend}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle P_{\mathrm{comb}}=\sigma\!\left((1-\omega)\,\mathrm{logit}\,P_{\mathrm{stat}}+\omega\,\mathrm{logit}\,p_{\mathrm{ML}}\right),\qquad \omega=0.30\ /\ 0.15\ /\ 0.05\ \ (\mathrm{HIGH/MEDIUM/LOW})$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{HIGH\_RISK}\ (P_{\mathrm{comb}}\geq 0.8),\quad \mathrm{LOW\_RISK}\ (P_{\mathrm{comb}}\leq 0.1),\quad \mathrm{REVIEW\ otherwise}$}\par\smallskip
A shown assessment only: the case decision still comes from the Stage 1 gate.
\where{backend/src/workflow.ts}

\section{Stage 2 evaluation and continuous learning}
{\color{mute}ml/stage2\_math.py, ml/stage2\_lib.py and ml/train\_stage2.py.}

\subsection*{Average precision and ROC AUC}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{AP}=\sum_n(R_n-R_{n-1})\,P_n,\qquad \mathrm{AUC}=P(s_{\mathrm{front}}>s_{\mathrm{genuine}})$}\par\smallskip
Area under the precision-recall curve (the right ranking metric for rare positives) and the probability a random front outranks a random genuine merchant.
\where{ml/stage2\_lib.py}

\subsection*{Brier score and Murphy decomposition}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{BS}=\frac{1}{N}\sum_i(p_i-y_i)^2=\mathrm{REL}-\mathrm{RES}+\mathrm{UNC}$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{REL}=\frac{1}{N}\sum_kn_k(\bar{p}_k-\bar{o}_k)^2,\qquad \mathrm{RES}=\frac{1}{N}\sum_kn_k(\bar{o}_k-\bar{o})^2,\qquad \mathrm{UNC}=\bar{o}(1-\bar{o})$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{BSS}=1-\frac{\mathrm{BS}}{\bar{o}(1-\bar{o})}$}\par\smallskip
Murphy 1973; exact when forecasts are constant within each of the 10 bins.
\where{ml/stage2\_math.py}

\subsection*{Expected Calibration Error}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{ECE}=\sum_{k=1}^{10}\frac{n_k}{N}\left|\bar{o}_k-\bar{p}_k\right|$}\par\smallskip
Guo et al. 2017, equal-width bins.
\where{ml/stage2\_lib.py, scripts/benchmark-10k.ts}

\subsection*{Recall, false-alarm rate, precision}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{recall}=\frac{TP}{TP+FN},\quad \mathrm{FAR}=\frac{FP}{FP+TN},\quad \mathrm{precision}=\frac{TP}{\max(1,TP+FP)}$}\par\smallskip
\where{ml/stage2\_lib.py, scripts/benchmark-10k.ts}

\subsection*{DeLong test for two correlated AUCs (DeLong et al. 1988)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle V_{10}(x_i)=\frac{1}{n}\sum_j\psi(x_i,y_j),\quad V_{01}(y_j)=\frac{1}{m}\sum_i\psi(x_i,y_j),\quad \psi=\mathbb{1}[x>y]+\frac{1}{2}\mathbb{1}[x=y]$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{Var}=\frac{S_{10}}{m}+\frac{S_{01}}{n},\qquad z=\frac{\widehat{\mathrm{AUC}}_1-\widehat{\mathrm{AUC}}_2}{\sqrt{\mathrm{Var}_{11}+\mathrm{Var}_{22}-2\,\mathrm{Var}_{12}}},\qquad p=2(1-\Phi(|z|))$}\par\smallskip
Computed in O(n log n) with midranks (Sun and Xu 2014).
\where{ml/stage2\_math.py}

\subsection*{Paired bootstrap difference}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \Delta^{*}_b=\mathrm{metric}(y^{*}_b,p^{\mathrm{new}*}_b)-\mathrm{metric}(y^{*}_b,p^{\mathrm{old}*}_b),\qquad \mathrm{CI}=[Q_{0.025},Q_{0.975}],\ \ P(\mathrm{worse})=\overline{\mathbb{1}[\Delta^{*}<0]}$}\par\smallskip
2,000 resamples of the same merchants for both models.
\where{ml/stage2\_lib.py}

\subsection*{Champion/challenger promotion rule}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{promote}\Leftrightarrow \mathrm{lo}(\Delta\mathrm{AP})\geq-0.01\ \wedge\ \mathrm{lo}(-\Delta\mathrm{Brier})\geq-0.01\ \wedge\ \mathrm{lo}(-\Delta\mathrm{LL})\geq-0.02$}\par\smallskip
Non-inferiority on three scores; the old model is kept for rollback and every decision is logged.
\where{ml/train\_stage2.py}

\subsection*{Nested cross-validation (Varma and Simon 2006)}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \hat{\mathrm{err}}=\frac{1}{N}\sum_{f=1}^{5}\sum_{i\in f}L\!\left(y_i,\;\mathcal{A}(\mathcal{D}\setminus f)(x_i)\right)$}\par\smallskip
Search, stacking and calibration are refitted inside every outer fold, so each merchant is scored once by a pipeline that never saw it.
\where{ml/train\_stage2.py}

\subsection*{Leave-one-fraud-type-out}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{recall}_k=\frac{\#\{p(x)\geq t:\ x\in k\}}{|k|},\qquad \mathrm{model\ trained\ on}\ \mathcal{D}\setminus k$}\par\smallskip
Generalisation to a fraud type never seen in training, with Wilson intervals.
\where{ml/train\_stage2.py}

\subsection*{Batch drift and fraud-rate estimation}
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle D=\max|F_{\mathrm{new}}-F_{\mathrm{ref}}|\ \ (\mathrm{KS\ per\ feature,\ BH\ q\ values}),\qquad \mathrm{PSI\ on\ score\ deciles}$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle p_i^\prime=\frac{(\pi/\pi_0)\,p_i}{(\pi/\pi_0)\,p_i+((1-\pi)/(1-\pi_0))(1-p_i)},\qquad \pi\leftarrow\frac{1}{n}\sum_ip_i^\prime\ \ \mathrm{(EM\ until\ |\Delta\pi|<10^{-8})}$}\par\smallskip
Saerens et al. 2002: estimates the real fraud rate of an unlabelled batch from scores calibrated at the training rate pi0.
\where{ml/serve.py, ml/stage2\_math.py}

\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle a^{(1)}=\mathrm{ReLU}(W_1\ast x+b_1)\odot m,\quad a^{(2)}=\mathrm{ReLU}(W_2\ast a^{(1)}+b_2)\odot m$}\par\smallskip
\par\smallskip\noindent\adjustbox{max width=\linewidth}{$\displaystyle \mathrm{GAP}=\frac{\sum_tm_ta^{(2)}_t}{\max(\sum_tm_t,1)},\quad \mathrm{GMP}=\max_{t:\,m_t=1}a^{(2)}_t,\quad p=\sigma\!\left(w^\top\mathrm{ReLU}(W_3[\mathrm{GAP};\mathrm{GMP};\mathrm{weekly}]+b_3)\right)$}\par\smallskip

\where{ml/train\_graph.py}
